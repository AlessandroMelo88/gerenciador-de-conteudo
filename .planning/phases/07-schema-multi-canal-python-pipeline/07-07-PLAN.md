youtube_oauth.py Phase 7 — CLI com --channel flag:
```python
# python -m src.youtube_oauth --channel futebol-em-cortes
# Gera /app/youtube/token-futebol-em-cortes.json

import argparse, json, os
from pathlib import Path
from google.oauth2.credentials import Credentials
from google_auth_oauthlib.flow import InstalledAppFlow

SCOPES = ['https://www.googleapis.com/auth/youtube.upload']
SECRETS_FILE = os.environ.get('YOUTUBE_CLIENT_SECRETS', '/app/youtube/client_secrets.json')

def main():
    parser = argparse.ArgumentParser(description='Gera token OAuth YouTube por canal-destino')
    parser.add_argument('--channel', required=True, help='Slug do canal destino (ex: futebol-em-cortes)')
    args = parser.parse_args()
    
    token_path = f'/app/youtube/token-{args.channel}.json'
    flow = InstalledAppFlow.from_client_secrets_file(SECRETS_FILE, SCOPES)
    creds = flow.run_local_server(port=0)
    
    token_data = json.loads(creds.to_json())
    Path(token_path).parent.mkdir(parents=True, exist_ok=True)
    with open(token_path, 'w') as f:
        json.dump(token_data, f, indent=2)
    print(f'Token salvo em: {token_path}')

if __name__ == '__main__':
    main()
```
