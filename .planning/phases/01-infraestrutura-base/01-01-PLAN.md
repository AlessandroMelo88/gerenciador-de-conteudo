```yaml
  n8n:
    image: docker.n8n.io/n8nio/n8n:2.27.0
    container_name: n8n
    restart: unless-stopped
    ports:
      - "5678:5678"
    environment:
      - GENERIC_TIMEZONE=America/Sao_Paulo
      - TZ=America/Sao_Paulo
      - N8N_ENCRYPTION_KEY=${N8N_ENCRYPTION_KEY}
    volumes:
      - ./canaldecortes/n8n/data:/home/node/.n8n
    networks:
      - internal
    depends_on:
      - mysql

  clip-processor:
    build: ./canaldecortes/clip-processor
    container_name: clip-processor
    restart: unless-stopped
    environment:
      - ANTHROPIC_API_KEY=${ANTHROPIC_API_KEY}
      - MYSQL_HOST=mysql
      - MYSQL_DATABASE=clips_automation
      - MYSQL_USER=clips_user
      - MYSQL_PASSWORD=${CLIPS_DB_PASSWORD}
    volumes:
      - ./canaldecortes/youtube/token.json:/app/token.json:ro
      - /tmp/clips:/tmp/clips
    networks:
      - internal
    depends_on:
      - whisper
      - mysql

  whisper:
    image: fedirz/faster-whisper-server:latest-cpu
    container_name: whisper
    restart: unless-stopped
    environment:
      - WHISPER_MODEL=Systran/faster-whisper-small
    volumes:
      - ./canaldecortes/whisper/models:/root/.cache/huggingface
    ports:
      - "8000:8000"
    networks:
      - internal
```

Nota: O arquivo usa `version: "3.8"` no topo — manter essa linha intacta.
Nota: clip-processor monta token.json como :ro (read-only). O arquivo pode não existir ainda — o container falhará se o arquivo não existir. Isso será resolvido no Plan 03 quando o token for gerado. Para Phase 1, adicionar um arquivo youtube/token.json placeholder vazio para o mount não falhar: criar /Users/alessandrobm1/develop/server/wordpress/canaldecortes/youtube/token.json com `{}`.
