```python
"""Rejeita um clip específico (CTRL-03).

Uso CLI: python -m src.rejeitar <clip_id>
Chamado pelo n8n via executeCommand no workflow 06-router.
"""
import os
import sys
from src.db import get_db_connection


def rejeitar(clip_id: int) -> int:
    """Marca o clip como rejected, remove o MP4 do disco, preserva raw video.

    Returns:
        0 em sucesso, 1 se clip não existe, 2 se status inválido.
    """
    conn = get_db_connection()
    try:
        with conn.cursor() as cur:
            cur.execute(
                "SELECT clip_path, status FROM generated_clips WHERE id=%s",
                (clip_id,)
            )
            row = cur.fetchone()
            if not row:
                print(f'ERRO: clip {clip_id} não existe')
                return 1
            if row['status'] not in ('pending', 'approved'):
                print(f'ERRO: clip {clip_id} está em status {row["status"]}')
                return 2

            cur.execute(
                "UPDATE generated_clips SET status='rejected' "
                "WHERE id=%s AND status IN ('pending', 'approved')",
                (clip_id,)
            )
            affected = cur.rowcount
        conn.commit()

        clip_path = row['clip_path']
        if affected and clip_path and os.path.exists(clip_path):
            os.remove(clip_path)
            print(f'OK: clip {clip_id} rejeitado, MP4 removido ({clip_path})')
        else:
            print(f'OK: clip {clip_id} rejeitado (MP4 não encontrado em {clip_path})')
        return 0
    finally:
        conn.close()


if __name__ == '__main__':
    if len(sys.argv) < 2:
        print('Uso: python -m src.rejeitar <clip_id>')
        sys.exit(2)
    try:
        clip_id = int(sys.argv[1])
    except ValueError:
        print(f'ERRO: clip_id inválido: {sys.argv[1]}')
        sys.exit(2)
    sys.exit(rejeitar(clip_id))
```

**Pontos importantes:**
1. **NÃO** apagar o raw video (source_videos.video_path ou equivalente). Apenas `clip_path` da tabela `generated_clips`.
2. Guard `status IN ('pending', 'approved')` permite rejeitar clip que já foi aprovado mas ainda não publicou — race com publisher (que faz `WHERE id=%s AND status='approved'` para mover para publishing): se rejeitar chegar primeiro, publisher pula silenciosamente.
3. `dict cursor` é o padrão do projeto — `row['clip_path']`, `row['status']`. Confirmado em db.py do projeto.
4. Connection ownership: quem abre fecha. Padrão Phase 2/5.
5. Exit code != 0 para falhas (n8n executeCommand vai propagar para o bot reportar erro).
