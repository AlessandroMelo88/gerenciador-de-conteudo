```python
"""Notificações proativas para o Telegram via webhook interno do n8n (CTRL-06).

Decisão: clip-processor não conhece TELEGRAM_BOT_TOKEN. Faz POST para um webhook
n8n interno (rede docker) que centraliza credencial e formatação das mensagens.

Eventos suportados (decisão de fase):
  - 'upload_published'  — upload bem-sucedido no YouTube
  - 'pipeline_failure'  — falha crítica em qualquer estágio
  - 'clip_ttl_warning'  — disparado por ttl_worker (já implementado direto lá)

Comportamento best-effort: nunca propaga exceções (não derruba o pipeline por
um problema de rede no notificador).
"""
import os
import requests


N8N_NOTIFY_URL = os.environ.get('N8N_NOTIFY_URL', 'http://n8n:5678/webhook/notify')


def notify(event_type: str, payload: dict, timeout: float = 5.0) -> bool:
    """Envia evento para o webhook n8n. Retorna True em sucesso, False em falha."""
    try:
        resp = requests.post(
            N8N_NOTIFY_URL,
            json={'event': event_type, 'payload': payload},
            timeout=timeout,
        )
        if resp.status_code >= 400:
            print(f'[NOTIFY] warn: n8n retornou HTTP {resp.status_code} para {event_type}')
            return False
        return True
    except requests.RequestException as exc:
        print(f'[NOTIFY] warn: falha ao notificar {event_type}: {exc}')
        return False
```
