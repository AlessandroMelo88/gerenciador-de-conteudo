    ```php
    <?php
    namespace App\Models;
    use Illuminate\Database\Eloquent\Model;
    use Illuminate\Database\Eloquent\Factories\HasFactory;
    use Illuminate\Support\Facades\Storage;
    class DestinationChannel extends Model
    {
        use HasFactory;
        protected $table = 'destination_channels';
        public $timestamps = true;
        protected $fillable = [
            'slug','name','niche','youtube_channel_id','credit_template','active','oauth_expired_flag',
        ];
        protected $casts = ['active' => 'bool', 'oauth_expired_flag' => 'bool'];

        /**
         * Regra:
         *   'expired'    if oauth_expired_flag == TRUE
         *   'authorized' if oauth_expired_flag == FALSE AND token file exists
         *   'missing'    otherwise
         * Contrato coberto por tests/Unit/DestinationChannelOauthStatusTest.php.
         */
        public function getOauthStatusAttribute(): string
        {
            if ($this->oauth_expired_flag) {
                return 'expired';
            }
            $exists = file_exists(config('services.clip_processor.token_dir', base_path('../youtube')) . "/token-{$this->slug}.json");
            return $exists ? 'authorized' : 'missing';
        }
    }
    ```
    NOTA sobre `token_dir`: o path final depende do bind mount. Adicionar em `config/services.php` (não fazer edit se já existe do Plan 08-01):

    ```php
        'token_dir' => env('CLIP_PROCESSOR_TOKEN_DIR', '/var/www/html/painel/../youtube'),
    ],
    ```
    O painel monta o volume `./canaldecortes/youtube` como `ro` — path final resolvido no runtime. Documentar no README que o operador pode override via `CLIP_PROCESSOR_TOKEN_DIR` no `.env`.
