    1. Criar `app/Console/Commands/CreatePainelUser.php`:
    ```php
    <?php
    namespace App\Console\Commands;
    use App\Models\User;
    use Illuminate\Console\Command;
    use Illuminate\Support\Facades\Hash;
    use function Laravel\Prompts\password;
    use function Laravel\Prompts\text;

    class CreatePainelUser extends Command
    {
        protected $signature = 'painel:create-user';
        protected $description = 'Cria o usuário operador do painel (interativo, senha nunca ecoa).';

        public function handle(): int
        {
            $email = text('Email do operador', required: true, validate: fn ($v) =>
                filter_var($v, FILTER_VALIDATE_EMAIL) ? null : 'Email inválido.'
            );

            if (User::query()->where('email', $email)->exists()) {
                $this->error("Já existe usuário com email {$email}. Use painel:reset-password para alterar a senha.");
                return self::FAILURE;
            }

            $pass = password('Senha (mínimo 10 caracteres, não será exibida)', required: true, validate: fn ($v) =>
                strlen($v) >= 10 ? null : 'Senha deve ter no mínimo 10 caracteres.'
            );

            User::create([
                'name' => 'Operador',
                'email' => $email,
                'password' => Hash::make($pass),
            ]);

            $this->info("Usuário {$email} criado com sucesso.");
            return self::SUCCESS;
        }
    }
    ```

    2. Criar `app/Console/Commands/ResetPainelPassword.php`:
    ```php
    <?php
    namespace App\Console\Commands;
    use App\Models\User;
    use Illuminate\Console\Command;
    use Illuminate\Support\Facades\Hash;
    use function Laravel\Prompts\password;

    class ResetPainelPassword extends Command
    {
        protected $signature = 'painel:reset-password {email}';
        protected $description = 'Reseta a senha do operador do painel.';

        public function handle(): int
        {
            $email = $this->argument('email');
            $user = User::query()->where('email', $email)->first();
            if (!$user) {
                $this->error("Nenhum usuário com email {$email}.");
                return self::FAILURE;
            }

            $pass = password('Nova senha (mínimo 10 caracteres, não será exibida)', required: true, validate: fn ($v) =>
                strlen($v) >= 10 ? null : 'Senha deve ter no mínimo 10 caracteres.'
            );

            $user->password = Hash::make($pass);
            $user->save();
            $this->info("Senha de {$email} atualizada.");
            return self::SUCCESS;
        }
    }
    ```
